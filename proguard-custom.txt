# cordova-plugin-proguard default rules
#
# Modern cordova-android builds use R8 through the Android Gradle Plugin.
# R8 accepts ProGuard-compatible rules, so this file keeps conservative,
# Cordova-focused defaults. Add project-specific rules in your app root
# `proguard-custom.txt`; the install hook appends them to this file.

# Keep metadata commonly needed by reflection, serializers, and stack traces.
-keepattributes *Annotation*,InnerClasses,EnclosingMethod,Signature,SourceFile,LineNumberTable
-renamesourcefileattribute SourceFile

# Keep enum helper methods.
-keepclassmembers enum * {
    public static **[] values();
    public static ** valueOf(java.lang.String);
}

# Keep generated Android resource classes.
-keep class **.R
-keep class **.R$* {
    <fields>;
}

# Keep Cordova core and plugin entry points.
-keep class org.apache.cordova.** { *; }
-keep public class * extends org.apache.cordova.CordovaPlugin

# Keep JS bridge methods annotated for WebView interfaces.
-keepclassmembers class * {
    @android.webkit.JavascriptInterface <methods>;
}

# Keep plugin manager references used by Cordova internals.
-keep class org.apache.cordova.CordovaBridge { org.apache.cordova.PluginManager pluginManager; }
-keep class org.apache.cordova.CordovaInterfaceImpl { org.apache.cordova.PluginManager pluginManager; }
-keep class org.apache.cordova.CordovaResourceApi { org.apache.cordova.PluginManager pluginManager; }
-keep class org.apache.cordova.CordovaWebViewImpl { org.apache.cordova.PluginManager pluginManager; }
-keep class org.apache.cordova.ResumeCallback { org.apache.cordova.PluginManager pluginManager; }
-keep class org.apache.cordova.engine.SystemWebViewEngine { org.apache.cordova.PluginManager pluginManager; }
